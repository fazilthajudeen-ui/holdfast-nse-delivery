% Preamble for main.tex (loaded after \documentclass). No package changes margins, spacing or font sizes.
% Fonts: Times (IEEEtran default) - under XeTeX via TeX Gyre Termes, the Times clone, since XeTeX cannot load
% the Type-1 Times; under pdfLaTeX via newtxtext. Math: newtxmath (Times-compatible).

\usepackage{cite}
\usepackage{amsmath,amssymb}
\usepackage{graphicx}
\usepackage{booktabs}
\usepackage{array}
\usepackage{url}

\usepackage{placeins}
\ifXeTeX
  \usepackage{newtxmath}
  \usepackage{fontspec}
  \setmainfont{texgyretermes}[Extension=.otf, UprightFont=*-regular, BoldFont=*-bold,
    ItalicFont=*-italic, BoldItalicFont=*-bolditalic]
  \setmonofont{texgyrecursor}[Extension=.otf, UprightFont=*-regular, BoldFont=*-bold,
    ItalicFont=*-italic, BoldItalicFont=*-bolditalic]
\else
  \usepackage[T1]{fontenc}
  \usepackage{newtxtext,newtxmath}
\fi
\usepackage{textcomp}
